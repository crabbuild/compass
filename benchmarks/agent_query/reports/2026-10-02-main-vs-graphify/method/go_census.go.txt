package main
import("encoding/json";"fmt";"go/ast";"go/parser";"go/token";"os";"path/filepath")
type Decl struct{File string `json:"file"`;Name string `json:"name"`;Line int `json:"line"`;Kind string `json:"kind"`;Owner string `json:"owner,omitempty"`}
func main(){root:=os.Args[1];rows:=[]Decl{};files:=[]string{};for _,rel:=range os.Args[2:]{set:=token.NewFileSet();f,e:=parser.ParseFile(set,filepath.Join(root,rel),nil,parser.AllErrors);if e!=nil{fmt.Fprintln(os.Stderr,e);os.Exit(1)};files=append(files,rel);for _,d:=range f.Decls{switch v:=d.(type){case *ast.FuncDecl:rows=append(rows,Decl{File:rel,Name:v.Name.Name,Line:set.Position(v.Pos()).Line,Kind:"function"});case *ast.GenDecl:if v.Tok!=token.TYPE{continue};for _,spec:=range v.Specs{t,ok:=spec.(*ast.TypeSpec);if !ok{continue};rows=append(rows,Decl{File:rel,Name:t.Name.Name,Line:set.Position(t.Pos()).Line,Kind:"type"});if s,ok:=t.Type.(*ast.StructType);ok{for _,field:=range s.Fields.List{for _,n:=range field.Names{if n.Name!="_"{rows=append(rows,Decl{File:rel,Name:n.Name,Line:set.Position(n.Pos()).Line,Kind:"field",Owner:t.Name.Name})}}}}}}}};json.NewEncoder(os.Stdout).Encode(map[string]any{"provider":"Go standard go/parser AST","files":files,"declarations":rows})}
