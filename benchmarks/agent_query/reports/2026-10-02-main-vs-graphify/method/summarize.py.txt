"""Publish a report only after every registered observation is present."""
import collections,hashlib,json,statistics
from pathlib import Path
from evaluate import ROOT,SUITES,SOURCES

def read(name):return json.loads((ROOT/name).read_text())
def med(xs):return statistics.median(xs)
def table(headers,rows):
    return '\n'.join(['| '+' | '.join(headers)+' |','| '+' | '.join(['---']*len(headers))+' |',*['| '+' | '.join(map(str,row))+' |' for row in rows]])
def variants():return ['compass','compass-sqlite','graphify','graphify-leiden']
LABEL={'compass':'Compass JSON','compass-sqlite':'Compass SQLite','graphify':'Graphify base','graphify-leiden':'Graphify native Leiden'}

def summarize():
    builds=read('builds.json');queries=read('queries.json');graphs=read('graphs.json');calls=read('source-call-audit.json');census=read('source-census-audit.json');directional=read('directional-query-audit.json');transitions=read('graph-transition-audit.json')
    assert len(builds)==120 and len(queries)==1020 and len(graphs)==20,'Incomplete run cannot publish a final report'
    assert len({(x['repository'],x['tool'],x['store'],x['trial'],x['mode']) for x in builds})==120
    assert len({(x['panel'],x['question'],x['variant'],x['trial']) for x in queries})==1020
    for entry in graphs:assert hashlib.sha256(Path(entry['graph']).read_bytes()).hexdigest()==entry['sha256']
    summary={'builds':[],'queries':[],'quality':{},'failures':[],'queryVariability':[],'health':[],'determinism':[],'directionalQueryAudit':directional['scores'],'matchedSuccessfulLatency':[],'graphTransitions':transitions['observations']}
    for repo in SOURCES:
        for v in variants():
            tool='compass' if v.startswith('compass') else v
            store='sqlite' if v=='compass-sqlite' else 'json'
            for mode in ('cold','unchanged'):
                x=[a for a in builds if a['repository']==repo and a['tool']==tool and a['store']==store and a['mode']==mode]
                summary['builds'].append(dict(repository=repo,variant=v,mode=mode,samples=len(x),
                    medianSeconds=med([a['wall_seconds'] for a in x]),rangeSeconds=[min(a['wall_seconds'] for a in x),max(a['wall_seconds'] for a in x)],
                    medianCpuSeconds=med([a['user_seconds']+a['system_seconds'] for a in x]),
                    medianChildPeakMiB=med([a['peak_rss_kib']/1024 for a in x]),medianSampledTreeMiB=med([a['sampledTreePeakRssKiB']/1024 for a in x]),
                    failures=sum(a['return_code']!=0 or a['timed_out'] for a in x)))
            x=[a for a in builds if a['repository']==repo and a['tool']==tool and a['store']==store]
            cold={a['trial']:a for a in x if a['mode']=='cold'};warm={a['trial']:a for a in x if a['mode']=='unchanged'}
            summary['determinism'].append(dict(repository=repo,variant=v,coldDistinctRawDigests=len({a['graphSha256'] for a in cold.values()}),
                unchangedMatches=[cold[i]['graphSha256']==warm[i]['graphSha256'] for i in range(3)]))
    for panel,suite in SUITES.items():
        for scope in ['all',*SOURCES]:
            for v in variants():
                x=[a for a in queries if a['panel']==panel and a['variant']==v and (scope=='all' or a['repository']==scope)]
                grouped={q:[a for a in x if a['question']==q] for q in sorted({a['question'] for a in x})}
                passed=sum(all(a['passed'] for a in obs) for obs in grouped.values())
                summary['queries'].append(dict(panel=panel,repository=scope,variant=v,questions=len(grouped),passedAllTrials=passed,
                    medianPerQuestionWallMs=med([med([a['wallMs'] for a in obs]) for obs in grouped.values()]),
                    medianFirstTrialWallMs=med([a['wallMs'] for a in x if a['trial']==0]),
                    meanTokens=statistics.mean([a['totalTokens'] for a in x]),timeouts=sum(a['timedOut'] for a in x),
                    sourceExcerptExcludedPassed=sum(all(a['passed'] for a in obs) for obs in grouped.values() if obs[0]['kind']!='explain_source'),
                    sourceExcerptExcludedQuestions=sum(obs[0]['kind']!='explain_source' for obs in grouped.values())))
                for q,obs in grouped.items():
                    if scope!='all':continue
                    if not all(a['passed'] for a in obs):summary['failures'].append(dict(panel=panel,question=q,repository=obs[0]['repository'],variant=v,passedTrials=sum(a['passed'] for a in obs),failures=[a['failures'] for a in obs]))
                    if len({a['passed'] for a in obs})>1:summary['queryVariability'].append(dict(panel=panel,question=q,variant=v))
    for panel in SUITES:
        candidates=sorted({a['question'] for a in queries if a['panel']==panel})
        shared=[]
        for q in candidates:
            obs=[a for a in queries if a['panel']==panel and a['question']==q]
            strict=[a for a in directional['observations'] if a['question']==q] if panel=='structured' else []
            if all(a['passed'] for a in obs) and all(a['passed'] for a in strict):shared.append(q)
        for v in variants():
            summary['matchedSuccessfulLatency'].append(dict(panel=panel,variant=v,questions=shared,medianWallMs=med([med([a['wallMs'] for a in queries if a['panel']==panel and a['variant']==v and a['question']==q]) for q in shared]) if shared else None))
    for v in ['compass','graphify','graphify-leiden']:
        pos=[a for a in calls if a['tool']==v and a['expected']=='present'];neg=[a for a in calls if a['tool']==v and a['expected']=='absent']
        summary['quality'][v]=dict(exactCalls=sum(a['exactOccurrence'] for a in pos),callTotal=len(pos),
            negativePass=sum(a['status']=='negative_pass' for a in neg),negativeTotal=len(neg),
            declarations=sum(a['matched'] for a in census if a['tool']==v),declarationTotal=sum(a['tool']==v for a in census))
    for e in graphs:
        g=json.loads(Path(e['graph']).read_text());meta=g.get('graph',{});ds=meta.get('diagnostics',[])
        summary['health'].append(dict(repository=e['repository'],variant=e['tool'],nodes=len(g['nodes']),edges=len(g.get('links',g.get('edges',[]))),
            graphMiB=Path(e['graph']).stat().st_size/1024**2,directed=g.get('directed'),multigraph=g.get('multigraph'),
            diagnostics=dict(collections.Counter(a.get('code') for a in ds)),
            publicationOmissions=[a.get('message','') for a in ds if a.get('code')=='publication_omission_summary'],
            fileStatuses=dict(collections.Counter(a.get('extractionStatus') for a in meta.get('files',[]))),
            extractedFilesByLanguage=dict(collections.Counter(a.get('language','unclassified') for a in meta.get('files',[]) if a.get('extractionStatus')=='extracted'))))
    (ROOT/'summary.json').write_text(json.dumps(summary,indent=2)+'\n')
    def qrow(panel,scope,v):return next(a for a in summary['queries'] if a['panel']==panel and a['repository']==scope and a['variant']==v)
    def brow(repo,v,mode):return next(a for a in summary['builds'] if a['repository']==repo and a['variant']==v and a['mode']==mode)
    def score(a):return f"{a['passedAllTrials']}/{a['questions']}"
    lines=['# Compass and Graphify evaluation on real repositories','',

Compass main is pinned at [b7b7fa819b127464566430cc454fd2085d5d5b62](https://github.com/crabbuild/compass/commit/b7b7fa819b127464566430cc454fd2085d5d5b62) (0.4.1); Graphify is the official [graphifyy 0.9.74 package](https://pypi.org/project/graphifyy/0.9.74/). This is a fresh measured comparison on the same read-only public source revisions, not a reuse of earlier performance claims.','',
        '**The results do not establish a universal winner.** Compass retains more of the independently checked declarations and exact call occurrences. Graphify builds the Zod corpus faster. Natural-language retrieval, supported output features, graph completeness, and storage mode need separate consideration; the tables below preserve those distinctions.','',
        '## Scope and fairness','',
        'The five language panels use Cobra (Go), Flask (Python), Gson (Java), Zod (TypeScript), and the Axum package (Rust). These are selected development repositories and existing source-reviewed questions. They are not a blinded or randomly sampled evaluation of all projects or languages.','',
        table(['Repository','Language','Source commit','Tracked files','Tracked MiB'],[(a['name'],a['language'],a['commit'][:12],a['trackedFiles'],f"{a['trackedBytes']/1024**2:.2f}") for a in read('registration.json')['repositories']]),'',
        'Both products use their code-only, community-clustered profiles without model credentials. Compass uses default low inference and is measured with JSON and SQLite storage separately; SQLite is its current default storage mode. Compass also inventories and can structurally extract non-code files. Graph size is therefore not an accuracy metric.','',
        table(['Reported fresh code scope','Compass headline file count','Graphify selected code files'],[(a['repository'],a['compassExtractedFiles'],a['graphifySelectedCodeFiles']) for a in read('file-scope-observations.json')['observations']]),'',
        'These headline counts agree for four corpora; Axum differs by one selected file.         'There are three fresh artifact roots and three unchanged rebuilds for each repository and configuration: 120 measured builds in total. Base Compass JSON and Graphify builds alternate order.         'Fresh builds mean fresh tool artifacts, with ordinary OS and runtime caches left in place. Unchanged rebuilds reuse the same output root.         '## Source checked graph quality','',
        'The call oracle was written from pinned source before graph generation.         table(['Graph producer','Exact call occurrences','Negative controls','Matched declarations in two censuses'],[(LABEL[v],f"{summary['quality'][v]['exactCalls']}/22",f"{summary['quality'][v]['negativePass']}/5",f"{summary['quality'][v]['declarations']}/827") for v in ['compass','graphify','graphify-leiden']]),'',
        table(['Repository','Compass exact calls','Graphify base exact calls','Graphify native Leiden exact calls'],[(repo,*[f"{sum(a['exactOccurrence'] for a in calls if a['repository']==repo and a['tool']==v and a['expected']=='present')}/{sum(a['repository']==repo and a['tool']==v and a['expected']=='present' for a in calls)}" for v in ['compass','graphify','graphify-leiden']]) for repo in SOURCES]),'',
        table(['Independent source census','Compass','Graphify base','Graphify native Leiden'],[(f"{repo} {kind}",*[f"{sum(a['matched'] for a in census if a['repository']==repo and a['tool']==v and a['declaration']['kind']==kind)}/{sum(a['repository']==repo and a['tool']==v and a['declaration']['kind']==kind for a in census)}" for v in ['compass','graphify','graphify-leiden']]) for repo,kind in [('flask','function'),('flask','class'),('cobra','function'),('cobra','type'),('cobra','field')]]),'',
        'The censuses cover all 24 `src/flask` Python files using CPython AST and all 19 tracked non-test Cobra Go files using Go’s standard parser.         'Concrete call gaps: Compass lacks `fromJSONSchema -> convertSchema` at Zod line 943.         'Compass’s published schema declares a directed multigraph. Repetitions are not independent accuracy examples. The ten ask tasks use identical text and a 2,000-token budget, reusing source facts from the structured panel.','',
        table(['Panel','Compass JSON','Compass SQLite','Graphify base','Graphify native Leiden'],[(panel,*[score(qrow(panel,'all',v)) for v in variants()]) for panel in SUITES]),'',

        table(['Structured panel excluding source-excerpt capability','Compass JSON','Compass SQLite','Graphify base','Graphify native Leiden'],[['45 questions',*[f"{qrow('structured','all',v)['sourceExcerptExcludedPassed']}/45" for v in variants()]]]),'',
        'A post-capture directional audit checks the ten caller/callee tasks against 19 requirements from 18 distinct facts in the original before-build source oracle. The Gson caller task now requires both reviewed overload calls, a stricter requirement than the original name-only score. Extra output remains unjudged.','',
        table(['Source-supported caller/callee requirements','Compass JSON','Compass SQLite','Graphify base','Graphify native Leiden'],[['10 tasks',*[f"{directional['scores'][v]['passedAllTrials']}/10" for v in variants()]]]),'',
        'Graphify’s Zod answer lists `convertBaseSchema` as an incoming caller at line 431 when the task asks for `convertSchema`’s callees. The original text scorer credits the name, but the reviewed outgoing call at line 814 is absent. Its Gson answer shows the real line-642 call, but omits line 724 and merges the overload declaration identities. Both native Leiden and base Graphify have these gaps; Compass answers all ten selected directional requirements. This does not establish complete answer precision.','',
        table(['Repository','Structured Compass / Graphify base','Natural query Compass / Graphify base','Ask Compass / Graphify base'],[(repo,*[score(qrow(p,repo,'compass'))+' / '+score(qrow(p,repo,'graphify')) for p in SUITES]) for repo in SOURCES]),'',
        'Compass fails two structured retrieval tasks on every trial: Cobra’s broad resolve/run question omits Find, and Axum’s bounded impact answer omits route_endpoint. Excluding the five source-excerpt capabilities leaves Graphify ahead on the original name-recall proxy, 44/45 versus Compass 43/45; the source-direction audit is a separate, stricter result.','',
        f"Timed-out query workflows: {sum(a['timedOut'] for a in queries)}. Questions with differing pass/fail results between trials: {len(summary['queryVariability'])}. These observations remain in the denominator. A timeout on the busy host does not by itself identify a root cause; successful retries do not erase the failed trial.",'',
        '## Indexing time and CPU cost','',
        'Times below are three-sample medians in seconds. No outlier was discarded. Ranges and all sample values remain in summary.json and builds.json. CPU time is accumulated user plus system time for the measured child workload and can exceed wall time when workers run in parallel.','',
